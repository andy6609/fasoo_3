# File-only AI upload interception policy
#
# Only browser traffic for ChatGPT, Gemini, and Claude is decrypted. The DLP
# engine separately ignores ordinary chat JSON, telemetry, and response bodies.
# Everything else uses a silent raw TLS tunnel and is never content-inspected.
#
# Format: <process> <target> <action> [reason]

# Local browser test server
chrome.exe demo.local:9443 MITM local_browser_test
chrome.exe 127.0.0.1:9443 MITM local_browser_test
chrome.exe localhost:9443 MITM local_browser_test
msedge.exe demo.local:9443 MITM local_browser_test
msedge.exe 127.0.0.1:9443 MITM local_browser_test
msedge.exe localhost:9443 MITM local_browser_test
firefox.exe demo.local:9443 MITM local_browser_test
firefox.exe 127.0.0.1:9443 MITM local_browser_test
firefox.exe localhost:9443 MITM local_browser_test

# ChatGPT / OpenAI
chrome.exe chatgpt.com:443 MITM ai_file_upload_only

chrome.exe *.oaiusercontent.com:443 MITM ai_file_upload_only
msedge.exe chatgpt.com:443 MITM ai_file_upload_only
msedge.exe *.openai.com:443 MITM ai_file_upload_only
msedge.exe *.oaiusercontent.com:443 MITM ai_file_upload_only
firefox.exe chatgpt.com:443 MITM ai_file_upload_only
firefox.exe *.openai.com:443 MITM ai_file_upload_only
firefox.exe *.oaiusercontent.com:443 MITM ai_file_upload_only
brave.exe chatgpt.com:443 MITM ai_file_upload_only
brave.exe *.openai.com:443 MITM ai_file_upload_only
brave.exe *.oaiusercontent.com:443 MITM ai_file_upload_only

# Fail-safe host-only fallback: process attribution can race with short-lived
# browser sockets. PAC routing is already AI-only, so an unknown process must
# not turn a configured AI upload into an encrypted bypass.
chatgpt.com:443 MITM ai_file_upload_process_fallback
*.chatgpt.com:443 MITM ai_file_upload_process_fallback
openai.com:443 MITM ai_file_upload_process_fallback
*.openai.com:443 MITM ai_file_upload_process_fallback
oaiusercontent.com:443 MITM ai_file_upload_process_fallback
*.oaiusercontent.com:443 MITM ai_file_upload_process_fallback
